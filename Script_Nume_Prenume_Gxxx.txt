-- Creare Tabele

-- 1. Sectii (Entitate Independenta)
CREATE TABLE Sectii (
    ID_Sectie NUMBER(4) PRIMARY KEY,
    Nume_Sectie VARCHAR2(50) NOT NULL,
    Etaj NUMBER(2)
);

-- 2. Cabinete (Entitate Dependenta de Sectii)
CREATE TABLE Cabinete (
    ID_Sectie NUMBER(4) NOT NULL,
    Numar_Cabinet NUMBER(3) NOT NULL,
    Descriere VARCHAR2(100),
    CONSTRAINT PK_Cabinete PRIMARY KEY (ID_Sectie, Numar_Cabinet),
    CONSTRAINT FK_Cabinete_Sectii FOREIGN KEY (ID_Sectie) REFERENCES Sectii(ID_Sectie) ON DELETE CASCADE
);

-- 3. Persoane (Super-type)
CREATE TABLE Persoane (
    CNP VARCHAR2(13) PRIMARY KEY,
    Nume VARCHAR2(50) NOT NULL,
    Prenume VARCHAR2(50) NOT NULL,
    Telefon VARCHAR2(15),
    Data_Nasterii DATE,
    CONSTRAINT CHK_CNP_Length CHECK (LENGTH(CNP) = 13)
);

-- 4. Medici (Sub-type din Persoane)
CREATE TABLE Medici (
    CNP VARCHAR2(13) PRIMARY KEY,
    Cod_Parafa VARCHAR2(10) UNIQUE NOT NULL,
    Specializare VARCHAR2(50) NOT NULL,
    ID_Sectie NUMBER(4),
    Numar_Cabinet NUMBER(3),
    CONSTRAINT FK_Medici_Persoane FOREIGN KEY (CNP) REFERENCES Persoane(CNP) ON DELETE CASCADE,
    CONSTRAINT FK_Medici_Cabinete FOREIGN KEY (ID_Sectie, Numar_Cabinet) REFERENCES Cabinete(ID_Sectie, Numar_Cabinet)
);

-- 5. Asiguratori (Pentru relatie 1:N cu Pacienti)
CREATE TABLE Asiguratori (
    ID_Asigurator NUMBER(4) PRIMARY KEY,
    Nume_Companie VARCHAR2(100) NOT NULL,
    Telefon_Contact VARCHAR2(15)
);

-- 6. Pacienti (Sub-type din Persoane)
CREATE TABLE Pacienti (
    CNP VARCHAR2(13) PRIMARY KEY,
    Grupa_Sanguina VARCHAR2(5),
    Alergii VARCHAR2(255) DEFAULT 'Nu se cunosc',
    ID_Asigurator NUMBER(4),
    CONSTRAINT FK_Pacienti_Persoane FOREIGN KEY (CNP) REFERENCES Persoane(CNP) ON DELETE CASCADE,
    CONSTRAINT FK_Pacienti_Asigurator FOREIGN KEY (ID_Asigurator) REFERENCES Asiguratori(ID_Asigurator)
);

-- 7. Dosare_Medicale (Relatie One-to-One cu Pacienti)
CREATE TABLE Dosare_Medicale (
    ID_Dosar NUMBER(6) PRIMARY KEY,
    CNP_Pacient VARCHAR2(13) UNIQUE NOT NULL,
    Data_Deschidere DATE DEFAULT SYSDATE,
    Istoric_General VARCHAR2(1000),
    CONSTRAINT FK_Dosare_Pacienti FOREIGN KEY (CNP_Pacient) REFERENCES Pacienti(CNP) ON DELETE CASCADE
);

-- 8. Servicii_Medicale
CREATE TABLE Servicii_Medicale (
    ID_Serviciu NUMBER(4) PRIMARY KEY,
    Denumire_Serviciu VARCHAR2(100) NOT NULL,
    Pret NUMBER(6, 2) CHECK (Pret >= 0),
    Durata_Minute NUMBER(3) CHECK (Durata_Minute > 0)
);

-- 9. Programari (Relație Ternară - Ordin 3: Medic, Pacient, Serviciu)
CREATE TABLE Programari (
    ID_Programare NUMBER(8) PRIMARY KEY,
    CNP_Medic VARCHAR2(13) NOT NULL,
    CNP_Pacient VARCHAR2(13) NOT NULL,
    ID_Serviciu NUMBER(4) NOT NULL,
    Data_Ora TIMESTAMP NOT NULL,
    Status VARCHAR2(20) DEFAULT 'Programat' CHECK (Status IN ('Programat', 'Finalizat', 'Anulat')),
    CONSTRAINT FK_Prog_Medici FOREIGN KEY (CNP_Medic) REFERENCES Medici(CNP),
    CONSTRAINT FK_Prog_Pacienti FOREIGN KEY (CNP_Pacient) REFERENCES Pacienti(CNP),
    CONSTRAINT FK_Prog_Servicii FOREIGN KEY (ID_Serviciu) REFERENCES Servicii_Medicale(ID_Serviciu),
    CONSTRAINT UQ_Programare_Medic UNIQUE (CNP_Medic, Data_Ora)
);

-- 10. Retete (Relatie 1:N de la Medic la Pacient)
CREATE TABLE Retete (
    ID_Reteta NUMBER(6) PRIMARY KEY,
    CNP_Medic VARCHAR2(13) NOT NULL,
    CNP_Pacient VARCHAR2(13) NOT NULL,
    Data_Eliberare DATE DEFAULT SYSDATE,
    Valabilitate_Zile NUMBER(3) DEFAULT 30,
    CONSTRAINT FK_Retete_Medici FOREIGN KEY (CNP_Medic) REFERENCES Medici(CNP),
    CONSTRAINT FK_Retete_Pacienti FOREIGN KEY (CNP_Pacient) REFERENCES Pacienti(CNP)
);

-- 11. Medicamente
CREATE TABLE Medicamente (
    ID_Medicament NUMBER(6) PRIMARY KEY,
    Denumire VARCHAR2(100) NOT NULL,
    Substanta_Activa VARCHAR2(100),
    Pret_Unitar NUMBER(6, 2) CHECK (Pret_Unitar > 0)
);

-- 12. Detalii_Reteta (Tabel Asociativ pentru M:N intre Retete si Medicamente)
CREATE TABLE Detalii_Reteta (
    ID_Reteta NUMBER(6) NOT NULL,
    ID_Medicament NUMBER(6) NOT NULL,
    Cantitate NUMBER(3) CHECK (Cantitate > 0),
    Dozaj VARCHAR2(50),
    CONSTRAINT PK_Detalii_Reteta PRIMARY KEY (ID_Reteta, ID_Medicament),
    CONSTRAINT FK_Detalii_Reteta FOREIGN KEY (ID_Reteta) REFERENCES Retete(ID_Reteta) ON DELETE CASCADE,
    CONSTRAINT FK_Detalii_Medicament FOREIGN KEY (ID_Medicament) REFERENCES Medicamente(ID_Medicament)
);
